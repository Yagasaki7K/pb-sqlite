# Integração com React

```tsx
'use client';

import { PbSqliteProvider, useQuery } from 'pb-sqlite/react';

export function AppShell({ children }: { children: React.ReactNode }) {
  return (
    <PbSqliteProvider dbName="minha-app" fallback={<p>Abrindo banco local…</p>}>
      {children}
    </PbSqliteProvider>
  );
}

function Lista() {
  const { data, isLoading, error, refetch } = useQuery<Nota>(
    'SELECT id, body FROM notes ORDER BY id DESC'
  );
  return isLoading ? <p>Carregando…</p> : <pre>{JSON.stringify(data)}</pre>;
}
```

`PbSqliteProvider` abre o banco, executa migrations e fecha a conexão ao desmontar. `usePbSqlite` acessa a instância e `useQuery` refaz consultas quando tabelas relevantes mudam.
